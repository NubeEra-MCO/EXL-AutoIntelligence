### `src/App.tsx`

```tsx
import { useEffect, useState } from "react";
import "./App.css";

import { EmployeeService } from "./generated/services/EmployeeService";
import type { Employee } from "./generated/models/EmployeeModel";

function App() {
  const [employees, setEmployees] = useState<Employee[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  const loadEmployees = async () => {
    try {
      setLoading(true);
      setError("");

      const result = await EmployeeService.getAll({
        top: 100,
      });

      setEmployees(result.data || []);
    } catch (err) {
      console.error("Failed to load employees:", err);
      setError(
        err instanceof Error
          ? err.message
          : "Failed to load employee records."
      );
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    loadEmployees();
  }, []);

  return (
    <div className="app">
      <header className="header">
        <div>
          <h1>Employee Management</h1>
          <p>Dataverse: crd3a_employee</p>
        </div>

        <button className="refresh-button" onClick={loadEmployees}>
          Refresh
        </button>
      </header>

      <main className="content">
        {loading && (
          <div className="status">
            Loading employees...
          </div>
        )}

        {!loading && error && (
          <div className="error">
            <strong>Error:</strong> {error}
          </div>
        )}

        {!loading && !error && (
          <>
            <div className="summary">
              <strong>{employees.length}</strong>
              <span>Employees</span>
            </div>

            {employees.length === 0 ? (
              <div className="status">
                No employee records found.
              </div>
            ) : (
              <div className="table-container">
                <table>
                  <thead>
                    <tr>
                      <th>Employee ID</th>
                      <th>Name</th>
                      <th>Email</th>
                      <th>Phone</th>
                      <th>Department</th>
                    </tr>
                  </thead>

                  <tbody>
                    {employees.map((employee) => (
                      <tr key={employee.crd3a_employeeid}>
                        <td>
                          {employee.crd3a_employeeid || "-"}
                        </td>

                        <td>
                          {employee.crd3a_name || "-"}
                        </td>

                        <td>
                          {employee.crd3a_email || "-"}
                        </td>

                        <td>
                          {employee.crd3a_phone || "-"}
                        </td>

                        <td>
                          {employee.crd3a_department || "-"}
                        </td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
            )}
          </>
        )}
      </main>
    </div>
  );
}

export default App;
```

### `generated/models/EmployeeModel.ts`

**Do not manually replace this file if PAC already generated it.** Keep your generated file because its exact properties depend on your Dataverse schema.

If your generated file has the following employee columns, it should effectively expose this shape:

```ts
export interface Employee {
  crd3a_employeeid: string;
  crd3a_name?: string | null;
  crd3a_email?: string | null;
  crd3a_phone?: string | null;
  crd3a_department?: string | null;
}
```

### `generated/services/EmployeeService.ts`

**Do not manually replace this file if PAC generated it.** Keep the generated service.

It needs to expose:

```ts
EmployeeService.getAll({
  top: 100
});
```

The generated Dataverse service supports `getAll()` for retrieving multiple records and accepts options such as `select`, `filter`, `orderBy`, `top`, and paging options. ([Microsoft Learn][1])
